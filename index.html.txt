        <div>规则：选出2个结果相等的卡片消除！</div>

    <button onclick="restartGame()">重新开始游戏</button>

<script>
// 卡片数据：{表达式,计算结果}
const cardList = [
    {q:"sin30°", val:0.5},
    {q:"cos60°", val:0.5},
    {q:"sin90°", val:1},
    {q:"tan45°", val:1},
    {q:"cos180°", val:-1},
    {q:"arccos(-1)", val:Math.PI},
    {q:"arcsin(0)", val:0},
    {q:"0", val:0},
    {q:"arcsin(1)", val:Math.PI/2},
    {q:"π/2", val:Math.PI/2},
    {q:"arctan(1)", val:Math.PI/4},
    {q:"π/4", val:Math.PI/4},
    {q:"cos0°", val:1},
    {q:"sin0°", val:0},
    {q:"arccos(1)", val:0},
    {q:"-1", val:-1},
];

let cards = [];
let selected = [];

// 打乱数组
function shuffle(arr){
    let a=[...arr];
    for(let i=a.length-1;i>0;i--){
        let j=Math.floor(Math.random()*(i+1));
        [a[i],a[j]] = [a[j],a[i]]
    }
    return a;
}

function initGame(){
    selected=[];
    cards = shuffle(cardList);
    const boardDom = document.getElementById("board");
    boardDom.innerHTML="";
    cards.forEach((item,idx)=>{
        const div = document.createElement("div");
        div.className="card";
        div.innerText=item.q;
        div.dataset.idx=idx;
        div.onclick = ()=>clickCard(idx);
        boardDom.appendChild(div);
    })
    updateRemain();
    document.getElementById("tip").innerText="请点击两张结果相等的卡片";
}

function clickCard(idx){
    if(selected.length>=2) return;
    const cardDom = document.querySelectorAll(".card")[idx];
    if(cardDom.classList.contains("match")) return;
    if(selected.includes(idx)) return;

    selected.push(idx);
    cardDom.classList.add("selected");

    if(selected.length===2){
        const i1=selected[0],i2=selected[1];
        const c1 = cards[i1], c2 = cards[i2];
        const dom1 = document.querySelectorAll(".card")[i1];
        const dom2 = document.querySelectorAll(".card")[i2];
        // 判断数值近似相等
        if(Math.abs(c1.val - c2.val) < 0.0001){
            //匹配成功
            document.getElementById("tip").innerText="✅匹配成功！消除！";
            dom1.classList.add("match");
            dom2.classList.add("match");
        }else{
            //匹配失败
            document.getElementById("tip").innerText="❌结果不相等！";
            dom1.classList.add("wrong");
            dom2.classList.add("wrong");
        }
        setTimeout(()=>{
            dom1.classList.remove("selected","wrong");
            dom2.classList.remove("selected","wrong");
            selected=[];
            updateRemain();
            checkWin();
        }, 800)
    }
}

function updateRemain(){
    let remain = document.querySelectorAll(".card:not(.match)").length;
    document.getElementById("remain").innerText = remain;
}

function checkWin(){
    const remain = document.querySelectorAll(".card:not(.match)").length;
    if(remain===0){
        document.getElementById("tip").innerText="🎉恭喜！全部消除通关！";
    }
}

function restartGame(){
    initGame();
}

initGame();
